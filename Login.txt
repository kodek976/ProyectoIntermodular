import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.control.*;
import javafx.scene.layout.*;
import javafx.scene.paint.Color;
import javafx.scene.text.*;
public class LoginView {
    private final StackPane root;
    // ── Colores globales ──────────────────────────────────────────────────────
    static final String BG         = "#0d0d0d";
    static final String CARD_BG    = "#1a1a1a";
    static final String ACCENT     = "#3b82f6";
    static final String TEXT_MAIN  = "#ffffff";
    static final String TEXT_SUB   = "#9ca3af";
    static final String FIELD_BG   = "#262626";
    static final String BORDER     = "#333333";
    public LoginView(Main app) {
        // ── Logo / brand ──────────────────────────────────────────────────────
        Label icon = new Label("⚙");
        icon.setFont(Font.font("System", 28));
        icon.setTextFill(Color.web(ACCENT));
        icon.setStyle(
            "-fx-background-color: #1e3a5f;" +
            "-fx-background-radius: 14;" +
            "-fx-padding: 14 16;"
        );
        Label brandTitle = new Label("YENAIAL");
        brandTitle.setFont(Font.font("System", FontWeight.BOLD, 22));
        brandTitle.setTextFill(Color.WHITE);
        Label brandSub = new Label("Taller de Bicicletas");
        brandSub.setFont(Font.font("System", 12));
        brandSub.setTextFill(Color.web(ACCENT));
        VBox brand = new VBox(4, icon, brandTitle, brandSub);
        brand.setAlignment(Pos.CENTER);
        // ── Card ──────────────────────────────────────────────────────────────
        Label heading = new Label("Iniciar Sesión");
        heading.setFont(Font.font("System", FontWeight.BOLD, 20));
        heading.setTextFill(Color.WHITE);
        // Email
        Label emailLabel = new Label("Correo Electrónico");
        emailLabel.setTextFill(Color.web(TEXT_SUB));
        emailLabel.setFont(Font.font("System", 12));
        TextField emailField = new TextField();
        emailField.setPromptText("tu@email.com");
        styleField(emailField);
        VBox emailBox = new VBox(5, emailLabel, emailField);
        // Contraseña
        Label passLabel = new Label("Contraseña");
        passLabel.setTextFill(Color.web(TEXT_SUB));
        passLabel.setFont(Font.font("System", 12));
        PasswordField passField = new PasswordField();
        passField.setPromptText("••••••••");
        styleField(passField);
        VBox passBox = new VBox(5, passLabel, passField);
        // Recordarme + olvidé
        CheckBox remember = new CheckBox("Recordarme");
        remember.setSelected(true);
        remember.setTextFill(Color.web(TEXT_SUB));
        remember.setStyle("-fx-mark-color: white;");
        Hyperlink forgot = new Hyperlink("¿Olvidaste tu contraseña?");
        forgot.setTextFill(Color.web(ACCENT));
        forgot.setBorder(Border.EMPTY);
        forgot.setFont(Font.font("System", 12));
        Region sp = new Region();
        HBox.setHgrow(sp, Priority.ALWAYS);
        HBox remRow = new HBox(remember, sp, forgot);
        remRow.setAlignment(Pos.CENTER_LEFT);
        // Botón login
        Button loginBtn = new Button("Iniciar Sesión");
        loginBtn.setMaxWidth(Double.MAX_VALUE);
        loginBtn.setPrefHeight(42);
        loginBtn.setFont(Font.font("System", FontWeight.BOLD, 14));
        stylePrimaryButton(loginBtn);
        loginBtn.setOnAction(e -> app.showDashboard());
        // Registro
        Label noAcc = new Label("¿No tienes una cuenta?");
        noAcc.setTextFill(Color.web(TEXT_SUB));
        noAcc.setFont(Font.font("System", 12));
        Hyperlink regLink = new Hyperlink("Regístrate");
        regLink.setTextFill(Color.web(ACCENT));
        regLink.setBorder(Border.EMPTY);
        regLink.setFont(Font.font("System", FontWeight.BOLD, 12));
        HBox regRow = new HBox(4, noAcc, regLink);
        regRow.setAlignment(Pos.CENTER);
        // Footer
        Label footer = new Label("© 2026 YENAIAL - Taller de Bicicletas. Todos los derechos reservados.");
        footer.setFont(Font.font("System", 10));
        footer.setTextFill(Color.web("#555555"));
        footer.setWrapText(true);
        footer.setTextAlignment(TextAlignment.CENTER);
        VBox card = new VBox(14,
            heading, emailBox, passBox, remRow, loginBtn, regRow, footer
        );
        card.setPadding(new Insets(32));
        card.setMaxWidth(420);
        card.setStyle(
            "-fx-background-color: " + CARD_BG + ";" +
            "-fx-background-radius: 12;" +
            "-fx-effect: dropshadow(gaussian, rgba(0,0,0,0.5), 20, 0, 0, 6);"
        );
        // ── Root ──────────────────────────────────────────────────────────────
        VBox center = new VBox(28, brand, card);
        center.setAlignment(Pos.CENTER);
        root = new StackPane(center);
        root.setStyle("-fx-background-color: " + BG + ";");
        StackPane.setAlignment(center, Pos.CENTER);
    }
    public StackPane getRoot() { return root; }
    // ── Helpers de estilo ─────────────────────────────────────────────────────
    static void styleField(TextField f) {
        f.setPrefHeight(40);
        f.setStyle(
            "-fx-background-color: " + FIELD_BG + ";" +
            "-fx-border-color: " + BORDER + ";" +
            "-fx-border-radius: 6;" +
            "-fx-background-radius: 6;" +
            "-fx-text-fill: white;" +
            "-fx-prompt-text-fill: #555555;" +
            "-fx-padding: 0 12;" +
            "-fx-font-size: 13px;"
        );
    }
    static void stylePrimaryButton(Button b) {
        String base =
            "-fx-background-color: " + ACCENT + ";" +
            "-fx-text-fill: white;" +
            "-fx-background-radius: 8;" +
            "-fx-cursor: hand;";
        String hover =
            "-fx-background-color: #2563eb;" +
            "-fx-text-fill: white;" +
            "-fx-background-radius: 8;" +
            "-fx-cursor: hand;";
        b.setStyle(base);
        b.setOnMouseEntered(e -> b.setStyle(hover));
        b.setOnMouseExited(e -> b.setStyle(base));
    }
}
