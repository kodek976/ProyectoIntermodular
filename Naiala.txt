<script>
    const API_KEY = "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";
    const MODEL   = "llama-3.1-8b-instant";

    const systemInstructionText = `
        Tu nombre es "Yenaial Bot". Eres un mecánico experto en ciclismo y el asistente virtual del "Taller de Bicicletas BiciFix".
        Tu objetivo es resolver dudas de mecánica, calcular presupuestos aproximados y guiar a los clientes de forma amable.
        Responde siempre de forma concisa y directa, sin alargarte innecesariamente.

        PRESENTACIÓN OBLIGATORIA:
        Si el usuario te pregunta quién eres, preséntate siempre como "Yenaial Bot, el asistente mecánico virtual de BiciFix".

        PRECIOS DE NUESTRA MANO DE OBRA (Presupuestos base):
        - Ajustar frenos o cambios: 25€.
        - Cambiar pastillas de freno o zapatas: 15€ (mano de obra por rueda).
        - Cambiar cadena o cassette: 20€.
        - Lavado y engrase completo: 15€.
        - Revisión integral (ajustes de seguridad, cables nuevos, centrado): 50€.
        - Reparación de pinchazos: 8€ (cámara estándar incluida).
        - Para otros problemas complejos (horquillas, pedalier), indica que cobramos la mano de obra a 35€/hora y calcula un estimado razonable.

        REGLAS PARA PRESUPUESTOS Y ENLACES DE REPUESTOS:
        1. Cuando te pidan un coste, desglosa el presupuesto: indica nuestro precio de mano de obra y aclara que el coste de las piezas va aparte. Da un total estimado combinando ambos factores.
        2. Si el usuario quiere comprar piezas, facilítale un enlace HTML así (nunca Markdown):
           <a href="https://amazon.es/s?k=PALABRAS+CLAVE" target="_blank">Ver [Nombre de la pieza] en Amazon</a>
        3. Recuerda al cliente que puede traer las piezas a BiciFix y nosotros nos encargamos de la instalación.
        4. Sé amable y mantén el foco en bicicletas. Si preguntan otra cosa, recházalo educadamente.
    `;

    const historial = [
        { role: "system", content: systemInstructionText }
    ];

    function formatearTexto(textoRAW) {
        if (!textoRAW) return "";
        let formateado = textoRAW.replace(/\n/g, '<br>');
        formateado = formateado.replace(/\*\*(.*?)\*\*/g, '<b>$1</b>');
        return formateado;
    }

    async function enviarPregunta() {
        const inputField = document.getElementById("userInput");
        const sendBtn    = document.getElementById("sendBtn");
        const chatBox    = document.getElementById("chatBox");
        const mensajeUsuario = inputField.value.trim();

        if (!mensajeUsuario) return;

        chatBox.innerHTML += `<div class="msg user-msg">${mensajeUsuario}</div>`;
        inputField.value = "";
        sendBtn.disabled = true;
        sendBtn.textContent = "...";
        chatBox.scrollTop = chatBox.scrollHeight;

        historial.push({ role: "user", content: mensajeUsuario });

        try {
            const response = await fetch("https://api.groq.com/openai/v1/chat/completions", {
                method: "POST",
                headers: {
                    "Authorization": `Bearer ${API_KEY}`,
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    model: MODEL,
                    messages: historial,
                    max_tokens: 400
                })
            });

            if (!response.ok) {
                const err = await response.json();
                throw new Error(JSON.stringify(err));
            }

            const data = await response.json();
            const respuestaIA = data.choices[0].message.content;

            historial.push({ role: "assistant", content: respuestaIA });
            chatBox.innerHTML += `<div class="msg ai-msg">${formatearTexto(respuestaIA)}</div>`;

        } catch (error) {
            console.error("Error:", error);
            historial.pop();
            chatBox.innerHTML += `<div class="msg error-msg"><b>Error:</b> No se pudo conectar con Yenaial Bot.</div>`;
        }

        sendBtn.disabled = false;
        sendBtn.textContent = "Enviar";
        chatBox.scrollTop = chatBox.scrollHeight;
    }

    document.getElementById("sendBtn").addEventListener("click", enviarPregunta);
    document.getElementById("userInput").addEventListener("keypress", function(e) {
        if (e.key === 'Enter') enviarPregunta();
    });
</script>
